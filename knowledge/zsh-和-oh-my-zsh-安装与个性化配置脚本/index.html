Zsh 和 Oh My Zsh 安装与个性化配置脚本

← 返回知识库

Linux / 系统#知识库#Linux / 系统

以下是一个完整的脚本，用于在 Ubuntu 系统上安装 Zsh、Oh My Zsh 并进行个性化配置。

#!/bin/bash

# 设置颜色输出
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m' # No Color

# 日志函数
log_info() {
    echo -e "${GREEN}[INFO]${NC} $(date '+%Y-%m-%d %H:%M:%S') - $1"
}

log_warning() {
    echo -e "${YELLOW}[WARNING]${NC} $(date '+%Y-%m-%d %H:%M:%S') - $1"
}

log_error() {
    echo -e "${RED}[ERROR]${NC} $(date '+%Y-%m-%d %H:%M:%S') - $1"
}

log_step() {
    echo -e "${BLUE}[STEP]${NC} $(date '+%Y-%m-%d %H:%M:%S') - $1"
}

# 检查是否以非root权限运行
check_not_root() {
    if [[ $EUID -eq 0 ]]; then
        log_error "此脚本不应以root权限运行，请以普通用户运行"
        exit 1
    fi
}

# 安装 Zsh
install_zsh() {
    log_step "安装 Zsh..."
    sudo apt-get update
    sudo apt-get install -y zsh curl git
    log_info "Zsh 安装完成"
}

# 安装 Oh My Zsh
install_oh_my_zsh() {
    log_step "安装 Oh My Zsh..."
    
    # 备份现有的 .zshrc 文件（如果存在）
    if [ -f "$HOME/.zshrc" ]; then
        log_info "备份现有的 .zshrc 文件"
        mv "$HOME/.zshrc" "$HOME/.zshrc.backup.$(date +%Y%m%d%H%M%S)"
    fi
    
    # 使用官方安装脚本安装 Oh My Zsh
    sh -c "$(curl -fsSL https://raw.githubusercontent.com/ohmyzsh/ohmyzsh/master/tools/install.sh)" "" --unattended
    
    if [ $? -eq 0 ]; then
        log_info "Oh My Zsh 安装完成"
    else
        log_error "Oh My Zsh 安装失败"
        exit 1
    fi
}

# 安装 Powerlevel10k 主题
install_powerlevel10k() {
    log_step "安装 Powerlevel10k 主题..."
    
    # 克隆主题仓库
    git clone --depth=1 https://github.com/romkatv/powerlevel10k.git ${ZSH_CUSTOM:-$HOME/.oh-my-zsh/custom}/themes/powerlevel10k
    
    # 设置主题
    sed -i 's/plugins=(.*)/plugins=(git zsh-autosuggestions zsh-syntax-highlighting zsh-completions)/' ~/.zshrc
    
    log_info "常用插件安装完成"
}

# 自定义配置
custom_configuration() {
    log_step "应用自定义配置..."
    
    # 添加常用别名
    cat >> ~/.zshrc << 'EOF'

# 自定义别名
alias ll='ls -alF'
alias la='ls -A'
alias l='ls -CF'
alias update='sudo apt update && sudo apt upgrade -y'
alias cls='clear'
alias zshconfig='nano ~/.zshrc'
alias ohmyzsh='nano ~/.oh-my-zsh'
alias grep='grep --color=auto'
alias egrep='egrep --color=auto'
alias fgrep='fgrep --color=auto'

# 历史命令配置
HIST_STAMPS="yyyy-mm-dd"
HISTSIZE=100000
SAVEHIST=100000
setopt HIST_IGNORE_ALL_DUPS
setopt HIST_SAVE_NO_DUPS
setopt HIST_REDUCE_BLANKS
setopt INC_APPEND_HISTORY_TIME
setopt EXTENDED_HISTORY

# 其他配置
setopt autocd              # 自动进入目录
setopt interactivecomments # 允许在交互模式中使用注释
setopt magicequalsubst     # 启用等号扩展
setopt nonomatch           # 如果没有匹配，不报错
setopt notify              # 显示后台任务完成状态
setopt numericglobsort     # 数字排序
setopt promptsubst         # 启用提示符扩展

# 提高命令补全
autoload -Uz compinit
compinit

EOF
    
    log_info "自定义配置完成"
}

# 设置为默认shell
set_default_shell() {
    log_step "设置 Zsh 为默认 shell..."
    
    # 检查当前shell
    CURRENT_SHELL=$(basename "$SHELL")
    if [ "$CURRENT_SHELL" != "zsh" ]; then
        # 更改默认shell
        chsh -s $(which zsh)
        log_info "Zsh 已设置为默认 shell，重新登录后生效"
    else
        log_info "Zsh 已经是默认 shell"
    fi
}

# 完成提示
completion_message() {
    echo ""
    log_info "Zsh 和 Oh My Zsh 安装配置完成！"
    log_info "要应用所有更改，请执行: source ~/.zshrc"
    log_info "或者重新启动终端"
    echo ""
    log_info "要配置 Powerlevel10k 主题，请重新启动终端后运行: p10k configure"
    echo ""
}

# 主函数
main() {
    log_step "开始安装和配置 Zsh 与 Oh My Zsh..."
    check_not_root
    install_zsh
    install_oh_my_zsh
    install_powerlevel10k
    install_plugins
    custom_configuration
    set_default_shell
    completion_message
}

# 执行主函数
main "$@"

使用说明

  1. 保存脚本：nano install_zsh.sh # 将上述脚本内容粘贴进去，然后按 Ctrl+X，输入 Y 保存
  2. 赋予执行权限：chmod +x install_zsh.sh
  3. 运行脚本：./install_zsh.sh

个性化配置选项

安装完成后，你可以进行以下个性化配置：

配置 Powerlevel10k 主题

重新启动终端后，运行以下命令配置 Powerlevel10k：

p10k configure

按照提示选择你喜欢的样式和图标。

添加更多插件

你可以在 ~/.zshrc 文件中添加更多插件。常用的插件包括：

  • autojump: 快速跳转到常用目录sudo apt install autojump 然后在插件列表中添加 autojump
  • colored-man-pages: 彩色man页面gi
  • ${ZSH_CUSTOM:-~/.oh-my-zsh/custom}/plugins/colored-man-pages 然后在插件列表中添加 colored-man-pages

自定义提示符

你可以在 ~/.zshrc 文件中添加自定义提示符设置。例如：

# 在 ~/.zshrc 中添加
PROMPT='%F{blue}%n%f@%F{green}%m%f:%F{yellow}%~%f$ '

添加更多别名

在 ~/.zshrc 文件中添加你常用的别名：

# 自定义别名
alias myip='curl http://ipecho.net/plain; echo'
alias weather='curl wttr.in'
alias ports='netstat -tulanp'

手动安装 Oh My Zsh 插件

如果你想手动安装其他插件，可以使用以下命令：

# 进入 Oh My Zsh 自定义插件目录
cd ${ZSH_CUSTOM:-~/.oh-my-zsh/custom}/plugins

# 克隆插件仓库
git clone <插件仓库URL>

# 编辑 ~/.zshrc 文件，将插件添加到插件列表
nano ~/.zshrc
# 找到 plugins=(...) 行，添加你的插件名称

恢复备份

如果安装过程中出现问题，你可以恢复备份的配置文件：

# 恢复 .zshrc 备份
mv ~/.zshrc.backup.* ~/.zshrc

# 如果你想完全卸载 Oh My Zsh
uninstall_oh_my_zsh

这个脚本提供了完整的 Zsh 和 Oh My Zsh 安装与配置流程，包括流行的 Powerlevel10k 主题和常用插件。你可以根据自己的喜好进一步自定义配置。
